<RunnablePlot title="Lab evaluasi statistik deskriptif" code={`import matplotlib
matplotlib.use("AGG")
from statistics import mean, median
import numpy as np
import matplotlib.pyplot as plt

waktu = [8, 10, 12, 12, 14, 16, 18, 30]
q1, q3 = np.quantile(waktu, [0.25, 0.75], method="linear")
iqr = q3 - q1
batas_bawah = q1 - 1.5 * iqr
batas_atas = q3 + 1.5 * iqr
kandidat = [x for x in waktu if x < batas_bawah or x > batas_atas]

print("n:", len(waktu))
print("Mean:", round(mean(waktu), 2))
print("Median:", median(waktu))
print("Jangkauan:", max(waktu) - min(waktu))
print("Q1, Q3, IQR:", round(q1, 2), round(q3, 2), round(iqr, 2))
print("Pagar atas:", round(batas_atas, 2))
print("Kandidat untuk diperiksa:", kandidat)

fig, ax = plt.subplots(figsize=(7, 4))
ax.boxplot(waktu, vert=False)
ax.scatter(waktu, [1] * len(waktu), alpha=0.45, color="#1876b8",
           label="Pengamatan")
ax.set_yticks([])
ax.set_xlabel("Waktu (menit)")
ax.set_title("Boxplot dan nilai pengamatan")
ax.legend()
fig.tight_layout()
fig.savefig("/tmp/statistik-plot.svg")
`} />
